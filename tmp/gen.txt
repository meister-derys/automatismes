const MINUS='−', NB=' ', NN=' ';
let R=Math.random;
function mulberry(seed){let a=seed>>>0;return function(){a=(a+0x6D2B79F5)|0;let t=Math.imul(a^(a>>>15),1|a);t=(t+Math.imul(t^(t>>>7),61|t))^t;return((t^(t>>>14))>>>0)/4294967296}}
const ri=(a,b)=>a+Math.floor(R()*(b-a+1));
const pick=a=>a[Math.floor(R()*a.length)];
const shuffle=a=>{a=a.slice();for(let i=a.length-1;i>0;i--){const j=Math.floor(R()*(i+1));[a[i],a[j]]=[a[j],a[i]]}return a};
const gcd=(a,b)=>{a=Math.abs(a);b=Math.abs(b);while(b){[a,b]=[b,a%b]}return a};
const lcm=(a,b)=>a/gcd(a,b)*b;

function num(x){
  const r=Math.round(x*1e4)/1e4, neg=r<0;
  let [i,f]=Math.abs(r).toString().split('.');
  if(i.length>=5) i=i.replace(/\B(?=(\d{3})+(?!\d))/g,NN);
  return (neg?MINUS:'')+i+(f?','+f:'');
}
const pct=(x,signed)=>(x<0?MINUS:(signed&&x>0?'+':''))+num(Math.abs(x))+NB+'%';
const eur=x=>num(x)+NB+'€';
const fr=(n,d)=>`<span class="fr"><span class="n">${n}</span><span class="d">${d}</span></span>`;
const ex=e=>e<0?MINUS+Math.abs(e):String(e);
const pw=(b,e)=>e===1?`${b}`:`${b}<sup>${ex(e)}</sup>`;
const sci=(m,e)=>`${num(m)}${NB}×${NB}10<sup>${ex(e)}</sup>`;
function rat(n,d){const g=gcd(n,d)||1;n/=g;d/=g;if(d<0){n=-n;d=-d}if(d===1)return num(n);return (n<0?MINUS:'')+fr(Math.abs(n),d)}
const eqRed=(n,d)=>{const g=gcd(n,d)||1;return g===1?fr(n,d):`${fr(n,d)} = ${rat(n,d)}`};
const o=(h,v)=>({h,v});
const ratO=(n,d)=>o(rat(n,d),n/d);
const okNums=(arr,c,min=0,max=Infinity)=>arr.filter(x=>Number.isFinite(x)&&x>min&&x<=max&&Math.abs(x-c)>1e-9);

function Q(prompt,correct,wrongs,explain,extra={}){
  const key=x=>typeof x.v==='number'?'n'+Number(x.v.toPrecision(9)):'s'+(x.v!==undefined?x.v:x.h);
  const seen=new Set([key(correct)]),opts=[];
  for(const w of wrongs){if(!w)continue;const k=key(w);if(seen.has(k))continue;seen.add(k);opts.push(w)}
  if(opts.length<3)throw new Error('distracteurs insuffisants');
  const all=shuffle([correct,...opts.slice(0,3)]);
  return {prompt,choices:all.map(x=>x.h),answer:all.indexOf(correct),explain,fig:extra.fig||'',_keys:all.map(key),ref:{v:correct.v},bac:extra.bac||null};
}
const Qfixed=(prompt,choices,answer,explain,extra={})=>({prompt,choices,answer,explain,fig:extra.fig||'',_keys:choices,ref:null,bac:extra.bac||null});
const nQ=(prompt,c,ws,fmt,explain,extra)=>Q(prompt,o(fmt(c),c),ws.filter(Number.isFinite).map(w=>o(fmt(w),w)),explain,extra);

const GENVER=1,KLASSES=['Première STMG','Première ST2S','Autre classe'];

function encodeResult({topic,seed,mode,answers,klass,name}){
  const ti=TOPICS.findIndex(t=>t.id===topic);
  const a=answers.map(x=>x===null?'-':String(x)).join('');

}
function decodeResult(code){
  const m=String(code).trim().match(/^AUTO(\d+)\.([0-9a-f]+)\.([0-9a-z]+)$/i);
  if(!m)return {ok:false,err:'Format de code inconnu'};
  if(+m[1]!==GENVER)return {ok:false,err:'Version de code différente'};
  const hex=m[2].toLowerCase();
  if(hex.length%2)return {ok:false,err:'Code incomplet'};
  let payload;try{payload=hexD(hex)}catch(e){return {ok:false,err:'Code illisible'}}
  if(h32('derys'+payload)!==m[3].toLowerCase())return {ok:false,err:'Code modifié ou incomplet'};
  const [ti,seed,mode,a,w,ki,...nm]=payload.split('|');
  const t=TOPICS[+ti];
  if(!t||!/^\d{3,6}$/.test(seed)||!/^[0-3-]{10}$/.test(a))return {ok:false,err:'Code invalide'};
  return {ok:true,topic:t.id,topicName:t.name,seed:+seed,mode:mode==='c'?'controle':'entrainement',
    answers:a.split('').map(x=>x==='-'?null:+x),when:parseInt(w,36)*60000,klass:KLASSES[+ki]||'Autre classe',name:nm.join('|')};
}

}

function poly(c){
  const parts=[];
  for(let k=c.length-1;k>=0;k--){
    const a=c[k];if(!a)continue;const ab=Math.abs(a);
    const body=k===0?num(ab):(ab===1?'':num(ab))+'x'+(k>1?`<sup>${k}</sup>`:'');
    parts.push((parts.length?(a<0?` ${MINUS} `:' + '):(a<0?MINUS:''))+body);
  }
  return parts.length?parts.join(''):'0';
}

/* ---------- Courbes : interpolation monotone (Fritsch–Butland) par des points à coordonnées entières ---------- */
function hermite(xs,ys){
  const n=xs.length,h=[],d=[],m=new Array(n);
  for(let i=0;i<n-1;i++){h[i]=xs[i+1]-xs[i];d[i]=(ys[i+1]-ys[i])/h[i]}
  m[0]=d[0];m[n-1]=d[n-2];
  for(let i=1;i<n-1;i++)m[i]=d[i-1]*d[i]<=0?0:3*(h[i-1]+h[i])/((2*h[i]+h[i-1])/d[i-1]+(h[i]+2*h[i-1])/d[i]);
  return x=>{let i=0;while(i<n-2&&x>xs[i+1])i++;const t=(x-xs[i])/h[i],t2=t*t,t3=t2*t;
    return (2*t3-3*t2+1)*ys[i]+(t3-2*t2+t)*h[i]*m[i]+(-2*t3+3*t2)*ys[i+1]+(t3-t2)*h[i]*m[i+1]};
}
/** Courbe sur [a ; b] : extremums à abscisses entières, et pour chaque niveau demandé, des points d'ordonnée ce niveau à abscisses entières. */
function makeCurve(levels,nExt){
  levels=levels||[];
  for(let t=0;t<400;t++){
    const a=-ri(4,6),b=ri(4,6),c=nExt!=null?nExt:pick([1,2,2,3]);
    const pos=[];let ok=true;
    for(let k=0;k<c;k++){const lo=(pos.length?pos[pos.length-1]+3:a+2),hi=b-2-3*(c-1-k);if(lo>hi){ok=false;break}pos.push(ri(lo,Math.min(hi,lo+3)))}
    if(!ok)continue;
    const kx=[a,...pos,b],ky=[ri(-4,4)];let up=R()<.5;
    for(let k=1;k<kx.length;k++){const p=ky[k-1],lo=up?p+3:-5,hi=up?5:p-3;if(lo>hi){ok=false;break}ky.push(ri(lo,hi));up=!up}
    if(!ok||levels.some(L=>ky.indexOf(L)>=0))continue;
    // Courbe lisse g passant par les extremums, puis léger recalage horizontal (lisse et croissant) pour que
    // chaque passage au niveau demandé tombe sur une abscisse entière.
    const g=hermite(kx,ky),nx=[],ox=[],X=[],Y=[];
    for(let k=0;k<kx.length-1;k++){
      nx.push(kx[k]);ox.push(kx[k]);X.push(kx[k]);Y.push(ky[k]);
      const y0=ky[k],y1=ky[k+1],L=levels.filter(l=>(l-y0)*(l-y1)<0).sort((u,v)=>y1>y0?u-v:v-u);
      let last=kx[k];
      for(const l of L){let lo=kx[k],hi=kx[k+1];for(let it=0;it<60;it++){const mid=(lo+hi)/2;if((g(mid)-l)*(y0-l)>0)lo=mid;else hi=mid}
        const xs=(lo+hi)/2,xi=Math.round(xs);if(xi<=last||xi>=kx[k+1]||Math.abs(xi-xs)>0.2){ok=false;break}
        nx.push(xi);ox.push(xs);X.push(xi);Y.push(l);last=xi}
      if(!ok)break;
    }
    if(!ok)continue;
    nx.push(kx[kx.length-1]);ox.push(kx[kx.length-1]);X.push(kx[kx.length-1]);Y.push(ky[ky.length-1]);
    const psi=hermite(nx,ox);
    return {a,b,X,Y,kx,ky,f:x=>g(psi(x))};
  }
  throw new Error('courbe');
}

function bacClean(raw,allowX){
  let s=String(raw==null?'':raw).slice(0,160).toLowerCase();

  s=s.replace(/^\s*[a-z' ]{3,}:\s*/,'');

/** Analyse d'une expression : nombres, + − × ÷, puissances, parenthèses, %, et la lettre x si autorisée. */
function bacParse(s,allowX){
  const V=allowX===true?'x':(allowX||'');
  let i=0;const isD=c=>c>='0'&&c<='9',E=()=>{throw new Error('syntaxe')};
  function expr(){let a=term();while(s[i]==='+'||s[i]==='-'){const o=s[i++],b=term();a=[o,a,b]}return a}
  function term(){let a=unary();for(;;){const c=s[i];
    if(c==='*'||c==='/'){i++;a=[c,a,unary()]}
    else if(c==='('||(V&&c&&V.indexOf(c)>=0)||(isD(c)&&s[i-1]===')')){a=['*',a,pow()]}
    else break}return a}
  function unary(){if(s[i]==='-'){i++;return ['n',unary()]}if(s[i]==='+'){i++;return unary()}return pow()}
  function pow(){let a=post();if(s[i]==='^'){i++;a=['^',a,unary()]}return a}
  function post(){let a=atom();while(s[i]==='%'){i++;a=['%',a]}return a}
  function atom(){const c=s[i];
    if(c==='('){i++;const a=expr();if(s[i]!==')')E();i++;return a}
    if(V&&c&&V.indexOf(c)>=0){i++;return ['x',c]}
    const m=/^(\d+\.?\d*|\.\d+)/.exec(s.slice(i));if(!m)E();i+=m[0].length;return ['c',parseFloat(m[0])]}
  try{
    if(!s||s.length>120)return {ok:false};
    const t=expr();if(i!==s.length)return {ok:false};
    return {ok:true,t,pct:/%/.test(s)};
  }catch(e){return {ok:false}}
}
function bacEval(t,x){
  switch(t[0]){
    case 'c':return t[1];case 'x':return typeof x==='object'?x[t[1]]:x;case 'n':return -bacEval(t[1],x);
    case '+':return bacEval(t[1],x)+bacEval(t[2],x);case '-':return bacEval(t[1],x)-bacEval(t[2],x);
    case '*':return bacEval(t[1],x)*bacEval(t[2],x);case '/':return bacEval(t[1],x)/bacEval(t[2],x);
    case '%':return bacEval(t[1],x)/100;case '^':return Math.pow(bacEval(t[1],x),bacEval(t[2],x));
  }
  return NaN;
}
/** Valeur numérique d'une réponse tapée : {ok,val,pct,s}. */
function bacNum(raw){
  let s=bacClean(raw,false);
  if(s.indexOf('=')>=0){const parts=s.split('=').filter(Boolean);s=parts.length?parts[parts.length-1]:''}
  const p=bacParse(s,false);
  if(!p.ok)return {ok:false,s};
  const val=bacEval(p.t,0);
  if(!Number.isFinite(val))return {ok:false,s};
  return {ok:true,val,pct:p.pct,s};
}
const bacEq=(a,b)=>Math.abs(a-b)<=1e-9*Math.max(1,Math.abs(b));
const bacNear=(a,b)=>b!==0&&Math.abs(a-b)<=0.005*Math.abs(b);
const NOTE_FORME='Valeur juste, mais la forme demandée n’est pas respectée.';
const NOTE_INCONNU='Réponse non reconnue automatiquement.';

/** Vérifie la réponse tapée b.kind : renvoie {r:'ok'|'ko'|'verif'|'vide',note}. */
function bacCheck(b,input){
  const raw=String(input==null?'':input).trim();
  if(!raw)return {r:'vide'};
  const kind=b.kind;
  if(kind==='eq'){
    let s=bacClean(raw,true);
    s=s.replace(/^(y|f\(x\)|f)=/,'');
    if(s.indexOf('=')>=0){const q=s.split('=');if(q.length===2&&/^(y|f\(x\))$/.test(q[1]))s=q[0];else return {r:'verif',note:NOTE_INCONNU}}
    const p=bacParse(s,true);
    if(!p.ok)return {r:'verif',note:NOTE_INCONNU};
    for(const x of [0,1,2,-3,4]){const v=bacEval(p.t,x);if(!Number.isFinite(v))return {r:'verif',note:NOTE_INCONNU};if(!bacEq(v,b.n/b.d*x+b.p))return {r:'ko'}}
    return {r:'ok'};
  }
  if(kind==='poly'){
    const rw=b.var?raw.replace(/^[^=]*=/,'').replace(new RegExp(b.var,'g'),'x'):raw;
    let s=bacClean(rw,true).replace(/^[a-z](\(x\))?=/,'');
    const p=bacParse(s,true);
    if(!p.ok)return {r:'verif',note:NOTE_INCONNU};
    const f=x=>b.c.reduce((a,c,k)=>a+c*Math.pow(x,k),0);
    for(const x of [0,1,2,-3,0.5]){const v=bacEval(p.t,x);if(!Number.isFinite(v))return {r:'verif',note:NOTE_INCONNU};if(!bacEq(v,f(x)))return {r:'ko'}}
    if(/[()]/.test(s))return {r:'verif',note:'Expression juste, mais elle n’est pas développée.'};
    const terms=s.replace(/^[+-]/,'').split(/[+-]/).length;
    if(terms>Math.max(1,b.c.filter(c=>c!==0).length))return {r:'verif',note:'Expression juste, mais elle n’est pas réduite.'};
    return {r:'ok'};
  }
  if(kind==='expr'){
    let s=bacClean(raw,b.vars);
    const q=s.split('=');if(q.length===2)s=q[0].length<=2?q[1]:(q[1].length<=2?q[0]:'');else if(q.length>2)s='';
    const p=bacParse(s,b.vars),e=bacParse(b.t,b.vars);
    if(!p.ok||!e.ok)return {r:'verif',note:NOTE_INCONNU};
    for(let k=0;k<4;k++){const env={};Array.from(b.vars).forEach((c,j)=>{env[c]=1.3+0.7*j+0.9*k+0.31*j*k});
      const v=bacEval(p.t,env),w=bacEval(e.t,env);if(!Number.isFinite(v))return {r:'verif',note:NOTE_INCONNU};if(!bacEq(v,w))return {r:'ko'}}
    return {r:'ok'};
  }
  if(kind==='ineq'){
    let s=bacClean(raw,'x').replace(/≤|=<|<=/g,'{').replace(/≥|=>|>=/g,'}');
    let op='',val=null,m;
    const flip={'<':'>','>':'<','{':'}','}':'{'};
    if((m=/^x([<>{}])(.+)$/.exec(s))){op=m[1];val=bacNum(m[2])}
    else if((m=/^(.+?)([<>{}])x$/.exec(s))){op=flip[m[2]];val=bacNum(m[1])}
    else if((m=/^([\[\]])(.+?);\+(∞|inf.*|oo)\[$/.exec(s))){op=m[1]===']'?'>':'}';val=bacNum(m[2])}
    else if((m=/^\]-(∞|inf[a-z]*|oo);(.+?)([\[\]])$/.exec(s))){op=m[3]==='['?'<':'{';val=bacNum(m[2])}
    if(!op||!val||!val.ok)return {r:'verif',note:NOTE_INCONNU};
    const want={'<':'<','>':'>','<=':'{','>=':'}'}[b.op];
    return op===want&&bacEq(val.val,b.v)?{r:'ok'}:{r:'ko'};
  }
  if(kind==='set'&&/aucun|pas de sol|impossible|∅|vide|^\{\}$|^s=\{?\}?$/.test(deacc(raw.toLowerCase())))return b.vs.length?{r:'ko'}:{r:'ok'};
  if(kind==='set'){
    let s=bacClean(raw,'x').replace(/^s=/,'').replace(/[{}]/g,'').replace(/x=/g,'').replace(/et|ou/g,';');
    const parts=s.split(';').filter(Boolean).map(bacNum);
    if(!parts.length||parts.some(x=>!x.ok))return {r:'verif',note:NOTE_INCONNU};
    const vals=parts.map(x=>x.val).filter((v,i,a)=>a.findIndex(w=>bacEq(w,v))===i);
    if(vals.length!==b.vs.length)return vals.length<b.vs.length&&vals.every(v=>b.vs.some(w=>bacEq(v,w)))?{r:'ko',note:'Il manque une solution.'}:{r:'ko'};
    return vals.every(v=>b.vs.some(w=>bacEq(v,w)))?{r:'ok'}:{r:'ko'};
  }
  if(kind==='yn'){
    const s=deacc(raw.toLowerCase()).trim();
    const w=/^(oui|vrai|o$)/.test(s)?'oui':/^(non|faux|n$)/.test(s)?'non':'';
    if(!w)return {r:'verif',note:NOTE_INCONNU};
    return w===b.v?{r:'ok'}:{r:'ko'};
  }
  if(kind==='intv'){
    let s=deacc(raw.toLowerCase()).replace(/[−‒–—‐‑]/g,'-').replace(/\s+/g,'').replace(/(\d),(\d)/g,'$1.$2').replace(/^(x∈|xappartienta|s=)/,'');
    if(/aucun|∅|vide|pasdesol/.test(s))return b.iv.length?{r:'ko'}:{r:'ok'};
    const parts=s.split(/∪|u|ou|et/).filter(Boolean),got=[];
    for(const q of parts){const m=/^([\[\]])(.+?);(.+?)([\[\]])$/.exec(q);if(!m)return {r:'verif',note:NOTE_INCONNU};
      const l=bacNum(m[2]),r=bacNum(m[3]);if(!l.ok||!r.ok)return {r:'verif',note:NOTE_INCONNU};got.push({l:l.val,r:r.val,lc:m[1]==='[',rc:m[4]===']'})}
    got.sort((u,v)=>u.l-v.l);
    if(got.length!==b.iv.length)return {r:'ko'};
    const same=got.every((g,i)=>bacEq(g.l,b.iv[i].l)&&bacEq(g.r,b.iv[i].r));
    if(!same)return {r:'ko'};
    return got.every((g,i)=>g.lc===b.iv[i].lc&&g.rc===b.iv[i].rc)?{r:'ok'}:{r:'ko',note:'Les bornes sont justes, mais pas les crochets.'};
  }
  if(kind==='sgn'){
    const g=raw.replace(/[−‒–—‐‑]/g,'-').replace(/[^+-]/g,''),w=b.s.map(c=>c==='+'?'+':'-').join('');
    if(!g)return {r:'verif',note:NOTE_INCONNU};
    return g===w?{r:'ok'}:{r:'ko'};
  }
  if(kind==='fact'){
    const s=bacClean(raw,true).replace(/^[a-z](\(x\))?=/,'');
    const p=bacParse(s,true);
    if(!p.ok)return {r:'verif',note:NOTE_INCONNU};
    const f=x=>b.c.reduce((a,c,k)=>a+c*Math.pow(x,k),0);
    for(const x of [0,1,2,-3,0.5]){const v=bacEval(p.t,x);if(!Number.isFinite(v))return {r:'verif',note:NOTE_INCONNU};if(!bacEq(v,f(x)))return {r:'ko'}}
    let t=p.t;while(t[0]==='n')t=t[1];
    return t[0]==='*'||t[0]==='^'?{r:'ok'}:{r:'ko',note:'Expression égale, mais elle n’est pas factorisée.'};
  }
  if(kind==='ineqx'){
    let s=bacClean(raw,'xy').replace(/≤|=<|<=/g,'{').replace(/≥|=>|>=/g,'}');
    const flip={'<':'>','>':'<','{':'}','}':'{'};let op='',e='',m;
    if((m=/^x([<>{}])(.+)$/.exec(s))){op=m[1];e=m[2]}
    else if((m=/^(.+?)([<>{}])x$/.exec(s))){op=flip[m[2]];e=m[1]}
    const p=e?bacParse(e,'y'):{ok:false},w=bacParse(b.t,'y');
    if(!op||!p.ok||!w.ok)return {r:'verif',note:NOTE_INCONNU};
    for(const y of [0,1,2.5,-3,7]){const v=bacEval(p.t,{y}),u=bacEval(w.t,{y});if(!Number.isFinite(v))return {r:'verif',note:NOTE_INCONNU};if(!bacEq(v,u))return {r:'ko'}}
    return op==={'<':'<','>':'>','<=':'{','>=':'}'}[b.op]?{r:'ok'}:{r:'ko'};
  }
  if(kind==='sym'){
    const s=raw.replace(/\s/g,'');
    if(!/^[<>=]$/.test(s))return {r:'verif',note:NOTE_INCONNU};
    return s===b.v?{r:'ok'}:{r:'ko'};
  }
  if(kind==='near'){
    const n=bacNum(raw);
    if(!n.ok)return {r:'verif',note:NOTE_INCONNU};
    if(bacEq(n.val,b.v))return {r:'ok'};
    return Math.abs(n.val-b.exact)<=b.tol*Math.abs(b.exact)?{r:'verif',note:'Ordre de grandeur acceptable, à valider.'}:{r:'ko'};
  }
  if(kind==='pair'){
    const s=deacc(raw.toLowerCase()).replace(/[−–—]/g,'-').replace(/\s+/g,'');
    let m=/^\(?([^;()]+)[;|]([^;()]+)\)?$/.exec(s)||/^\((-?\d+),(-?\d+)\)$/.exec(s);
    if(!m)return {r:'verif',note:NOTE_INCONNU};
    const a=bacNum(m[1]),c=bacNum(m[2]);
    if(!a.ok||!c.ok)return {r:'verif',note:NOTE_INCONNU};
    return bacEq(b.n/b.d*a.val+b.p,c.val)?{r:'ok'}:{r:'ko'};
  }
  if(kind==='txt'){
    const s=deacc(raw.toLowerCase());
    const w=/decroissant/.test(s)?'decroissante':/croissant/.test(s)?'croissante':/constant/.test(s)?'constante':'';
    if(!w)return {r:'verif',note:NOTE_INCONNU};
    return w===b.v?{r:'ok'}:{r:'ko'};
  }
  const n=bacNum(raw);
  if(kind==='sci'){
    const e=b.m*Math.pow(10,b.e);
    const m=/^(-?\d+\.?\d*)\*10\^(-?\d+)$/.exec(n.s);
    if(m){const mant=parseFloat(m[1]),val=mant*Math.pow(10,parseInt(m[2],10));
      if(!bacEq(val,e))return {r:'ko'};
      return Math.abs(mant)>=1&&Math.abs(mant)<10?{r:'ok'}:{r:'verif',note:'Valeur juste, mais ce n’est pas une écriture scientifique (le facteur doit être compris entre 1 et 10).'};}
    if(!n.ok)return {r:'verif',note:NOTE_INCONNU};
    return bacEq(n.val,e)?{r:'verif',note:'Valeur juste, mais ce n’est pas l’écriture scientifique demandée.'}:{r:'ko'};
  }
  if(kind==='pow'){
    const e=Math.pow(b.a,b.e);
    const m=/^(\d+)\^(-?\d+)$/.exec(n.s);
    if(m){const bs=parseInt(m[1],10),ex=parseInt(m[2],10);
      if(bs===b.a&&ex===b.e)return {r:'ok'};
      return bacEq(Math.pow(bs,ex),e)?{r:'verif',note:'Valeur juste, mais pas sous la forme d’une seule puissance de '+b.a+'.'}:{r:'ko'};}
    if(!n.ok)return {r:'verif',note:NOTE_INCONNU};
    return bacEq(n.val,e)?{r:'verif',note:'Valeur juste, mais pas sous la forme d’une seule puissance de '+b.a+'.'}:{r:'ko'};
  }
  if(!n.ok)return {r:'verif',note:NOTE_INCONNU};
  if(kind==='pct'){
    if(bacEq(n.val,b.v/100)||(!n.pct&&bacEq(n.val,b.v)))return {r:'ok'};
    if(bacNear(n.val,b.v/100)||(!n.pct&&bacNear(n.val,b.v)))return {r:'verif',note:'Valeur approchée.'};
    return {r:'ko'};
  }
  if(bacEq(n.val,b.v)){
    if(kind==='irred'){
      const m=/^-?(\d+)\/(\d+)$/.exec(n.s);
      if(m&&gcd(+m[1],+m[2])>1)return {r:'verif',note:'Valeur juste, mais la fraction n’est pas simplifiée.'};
    }
    if(kind==='dec'&&/[\/^]/.test(n.s))return {r:'verif',note:'Valeur juste, mais l’écriture décimale est demandée.'};
    return {r:'ok'};
  }
  if(bacNear(n.val,b.v))return {r:'verif',note:'Valeur approchée.'};
  return {r:'ko'};
}

/* ---------- Évaluations : questions d'une séance ---------- */
const FORMATS=['qcm','ecrit'];
const OLD5=['fractions','puissances','pourcentages','evolutions','droites'];
const BASE_TOPICS=AUTOS.map(a=>a.id).concat(OLD5,CALCS.map(c=>c.id));
function evalTopics(ids){const L=[];for(const id of (ids||[])){if(id==='bilan')L.push(...OLD5);else if(BASE_TOPICS.indexOf(id)>=0)L.push(id)}return L.filter((x,i)=>L.indexOf(x)===i)}
function evalMix(mix){const o={};let n=0;if(mix&&typeof mix==='object')for(const t of BASE_TOPICS){const c=Math.max(0,Math.min(20,parseInt(mix[t],10)||0));if(c){o[t]=c;n+=c}}return n?{mix:o,n:n}:null}
function evalSeedFor(cfg,name){return cfg.pseed?cfg.pseed:cfg.perso?100+parseInt(h32(cfg.seed+'|'+nameKey(name)),36)%899900:cfg.seed}
/** Questions d'un élève pour une séance : déterministe à partir de (sujets, nombre, graine, nom). */
function buildEval(cfg,name){
  const mx=evalMix(cfg.mix),topics=mx?Object.keys(mx.mix):evalTopics(cfg.topics),n=mx?mx.n:cfg.n,seed=evalSeedFor(cfg,name);
  if(!topics.length)throw new Error('aucun sujet');
  R=mulberry((seed*40503+n*7+topics.length*131)>>>0);
  try{
    const pools={},plan=[];
    const take=t=>{if(!pools[t]||!pools[t].length)pools[t]=shuffle(GENS[t]);plan.push(pools[t].pop())};
    if(mx){
      // mêmes types de questions pour toute la classe (tirés d'après la version commune)…
      R=mulberry((cfg.seed*40503+n*7+topics.length*131)>>>0);
      for(const t of topics)for(let k=0;k<mx.mix[t];k++)take(t);
      // …puis ordre et valeurs propres à chaque élève
      R=mulberry((seed*40503+n*7+topics.length*131+977)>>>0);
    }
    else for(let k=0;k<n;k++)take(topics[k%topics.length]);
    const qs=[],seen=new Set();
    for(const g of shuffle(plan)){
      let q=null;
      for(let k=0;k<80&&!q;k++){try{const c=g.run();const key=c.prompt+c.fig;if(!seen.has(key)){seen.add(key);q=c}}catch(e){}}
      if(!q)throw new Error('génération impossible : '+g.tag);
      q.tag=g.tag;
      if(cfg.fmt==='ecrit'){
        q.bac=q.bac||{kind:BACKIND[g.tag]||'num',v:q.ref?q.ref.v:0};
        if(q.bac.prompt)q.prompt=q.bac.prompt;
        if(q.bac.ans!=null){q.choices=[q.bac.ans];q.answer=0}
      }
      qs.push(q);
    }
    return qs;
  }finally{R=Math.random}
}
/** Corrige une copie. answers : indices (qcm) ou textes (ecrit). dec : décisions du professeur {numéro:true|false}. */
function gradeEval(cfg,name,answers,dec){
  const qs=buildEval(cfg,name);dec=dec||{};
  const rows=qs.map((q,k)=>{
    const g=answers[k];let r,note='';
    if(cfg.fmt==='qcm'){r=(g===null||g===undefined)?'vide':(g===q.answer?'ok':'ko')}
    else{const c=bacCheck(q.bac,g);r=c.r;note=c.note||''}
    let eff=r;if(r==='verif'){if(dec[k]===true)eff='ok';else if(dec[k]===false)eff='ko'}
    return {tag:q.tag,prompt:q.prompt,fig:q.fig,expected:q.choices[q.answer],explain:q.explain,given:g===undefined?null:g,r,eff,note,
      choices:cfg.fmt==='qcm'?q.choices:null,answer:q.answer,decided:dec[k]};
  });
  return {rows,score:rows.filter(x=>x.eff==='ok').length,total:rows.length,nVerif:rows.filter(x=>x.eff==='verif').length};
}
